@Service
public class WazuhTokenManager {

    private final WebClient webClient;
    private final String wazuhUsername;
    private final String wazuhPassword;

    // Cache
    private String cachedToken;
    private Instant tokenExpiryTime;

    // Le constructeur est aussi simplifié ici
    public WazuhTokenManager(
            @Qualifier("wazuhApiWebClient") WebClient wazuhWebClient, // <-- CORRECTION ICI
            @Value("${wazuh.api.username}") String wazuhUsername,
            @Value("${wazuh.api.password}") String wazuhPassword) {
        
        this.webClient = wazuhWebClient;
        this.wazuhUsername = wazuhUsername;
        this.wazuhPassword = wazuhPassword;
    }

    /**
     * Méthode publique principale. Retourne un token JWT valide,
     * en le rafraîchissant si nécessaire.
     * Cette méthode est synchronisée pour éviter les problèmes de concurrence.
     * @return Un token JWT valide.
     */
    public synchronized String getValidToken() {
        if (cachedToken == null || tokenExpiryTime == null || isTokenExpired()) {
            fetchNewToken();
        }
        return cachedToken;
    }

    /**
     * Méthode privée pour appeler l'API Wazuh et obtenir un nouveau token.
     */
    private void fetchNewToken() {
        System.out.println("Token expiré ou absent. Récupération d'un nouveau token...");
        try {
            WazuhAuthResponse response = this.webClient.post()
                    .uri("/security/user/authenticate")
                    .header(HttpHeaders.AUTHORIZATION, "Basic " + Base64.getEncoder().encodeToString((wazuhUsername + ":" + wazuhPassword).getBytes()))
                    .retrieve()
                    .bodyToMono(WazuhAuthResponse.class)
                    .block();

            if (response != null && response.getData() != null && response.getData().getToken() != null) {
                this.cachedToken = response.getData().getToken();
                this.tokenExpiryTime = parseExpiryFromToken(this.cachedToken);
                System.out.println("Nouveau token récupéré avec succès. Valide jusqu'à: " + this.tokenExpiryTime);
            } else {
                 throw new RuntimeException("La réponse d'authentification de Wazuh est invalide ou ne contient pas de token.");
            }
        } catch (Exception e) {
            System.err.println("Erreur lors de la récupération du token Wazuh: " + e.getMessage());
            throw new RuntimeException("Impossible de s'authentifier à l'API Wazuh", e);
        }
    }

    /**
     * Vérifie si le token en cache est expiré (avec une marge de sécurité).
     */
    private boolean isTokenExpired() {
        // On prend une marge de sécurité de 60 secondes pour éviter d'utiliser un token
        // qui expirerait entre sa vérification et son utilisation effective.
        return Instant.now().isAfter(this.tokenExpiryTime.minusSeconds(60));
    }

    /**
     * Parse le payload du JWT pour en extraire la date d'expiration ('exp').
     */
    private Instant parseExpiryFromToken(String token) {
        try {
            String[] parts = token.split("\\.");
            Base64.Decoder decoder = Base64.getUrlDecoder();
            String payload = new String(decoder.decode(parts[1]));

            ObjectMapper mapper = new ObjectMapper();
            @SuppressWarnings("unchecked")
            Map<String, Object> payloadMap = mapper.readValue(payload, Map.class);
            
            long expTimestamp = ((Number) payloadMap.get("exp")).longValue();
            return Instant.ofEpochSecond(expTimestamp);
        } catch (Exception e) {
            System.err.println("Impossible de parser la date d'expiration du token JWT. Expiration par défaut.");
            return Instant.now();
        }
    }

    private static class WazuhAuthResponse {
        private WazuhTokenData data;
        public WazuhTokenData getData() { return data; }
        public void setData(WazuhTokenData data) { this.data = data; }
    }

    private static class WazuhTokenData {
        private String token;
        public String getToken() { return token; }
        public void setToken(String token) { this.token = token; }
    }

@Service
public class WazuhAgentService {

    private final WebClient webClient;
    private final WazuhTokenManager tokenManager;
    private final UserRepository userRepository;
    private final AgentRepository agentRepository;
    private static final Duration API_TIMEOUT = Duration.ofSeconds(30);

    public WazuhAgentService(
            @Qualifier("wazuhApiWebClient") WebClient webClient,
            WazuhTokenManager tokenManager,
            UserRepository userRepository,
            AgentRepository agentRepository) {
        this.webClient = webClient;
        this.tokenManager = tokenManager;
        this.userRepository = userRepository;
        this.agentRepository = agentRepository;
    }

    /**
     * Méthode principale qui retourne une liste d'agents enrichie et filtrée
     * en fonction des permissions de l'utilisateur.
     *
     * @param username Le nom de l'utilisateur effectuant la requête.
     * @return Une liste de DTOs EnrichedAgentResponse.
     */
    public List<EnrichedAgentResponse> getAgentsForUser(String username) {
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new RuntimeException("Utilisateur non trouvé: " + username));

        // 1. Récupérer les données brutes de l'API Wazuh
        List<WazuhAgent> allAgentsFromWazuh = getAllAgentsFromWazuhAPI();
        
        // 2. Récupérer toutes nos données locales sur les agents
        List<Agent> allLocalAgents = agentRepository.findAll();
        Map<String, Agent> localAgentsMap = allLocalAgents.stream()
            .collect(Collectors.toMap(Agent::getId, agent -> agent));

        // 3. Fusionner les deux sources pour créer une liste d'agents enrichis
        List<EnrichedAgentResponse> allEnrichedAgents = allAgentsFromWazuh.stream()
            .map(wazuhAgent -> new EnrichedAgentResponse(wazuhAgent, localAgentsMap.get(wazuhAgent.getId())))
            .collect(Collectors.toList());

        // 4. Filtrer la liste enrichie en fonction du rôle de l'utilisateur
        Role userRole = user.getRole();
        switch (userRole) {
            case ROLE_SUPER_ADMIN:
                return allEnrichedAgents;

            case ROLE_ADMIN:
                if (user.getCustomerGroup() == null) return Collections.emptyList();
                Set<String> groupAgentIds = user.getCustomerGroup().getAgents().stream()
                        .map(Agent::getId).collect(Collectors.toSet());
                return allEnrichedAgents.stream()
                        .filter(agent -> groupAgentIds.contains(agent.getId()))
                        .collect(Collectors.toList());

            case ROLE_USER:
            default:
                Set<String> userAgentIds = user.getAgents().stream()
                        .map(Agent::getId).collect(Collectors.toSet());
                return allEnrichedAgents.stream()
                        .filter(agent -> userAgentIds.contains(agent.getId()))
                        .collect(Collectors.toList());
        }
    }

    /**
     * Méthode privée qui appelle l'API Wazuh pour obtenir la liste brute des agents.
     * @return Une liste de DTOs WazuhAgent non enrichis.
     */
    public List<WazuhAgent> getAllAgentsFromWazuhAPI() {
        try {
            String token = tokenManager.getValidToken();
            WazuhAgentResponse response = webClient.get()
                    .uri("/agents")
                    .header(HttpHeaders.AUTHORIZATION, "Bearer " + token)
                    .retrieve()
                    .bodyToMono(WazuhAgentResponse.class)
                    .timeout(API_TIMEOUT)
                    .block();

            if (response != null && response.getData() != null && response.getData().getAffectedItems() != null) {
                return response.getData().getAffectedItems();
            }
            return Collections.emptyList();
        } catch (Exception e) {
            System.err.println("Exception lors de la récupération des agents depuis Wazuh: " + e.getMessage());
            throw new RuntimeException("Erreur de communication avec l'API Wazuh", e);
        }
    }
    
    public Optional<WazuhAgent> getAgentById(String agentId) {
        try {
            String token = tokenManager.getValidToken();
            
            WazuhAgentResponse response = webClient.get()
            		.uri("/agents?agents_list={agentId}", agentId)
                    .header(HttpHeaders.AUTHORIZATION, "Bearer " + token)
                    .retrieve()
                    .bodyToMono(WazuhAgentResponse.class)
                    .timeout(API_TIMEOUT)
                    .block();

            if (response != null && response.getData() != null && response.getData().getAffectedItems() != null && !response.getData().getAffectedItems().isEmpty()) {
                // L'API retourne une liste, même pour un seul agent, donc on prend le premier élément.

            }
            
            return Optional.empty();
        } catch (Exception e) {
            System.err.println("Exception lors de la récupération de l'agent " + agentId + " depuis Wazuh: " + e.getMessage());
            return Optional.empty();
        }
    }

    private final WebClient indexerWebClient;
    private final OvalVulnerabilityFilter ovalFilter;

    public WazuhVulnerabilityService(@Qualifier("indexerWebClient") WebClient indexerWebClient,
                                     OvalVulnerabilityFilter ovalFilter) {
        this.indexerWebClient = indexerWebClient;

    }

    /**
     * Récupère les vulnérabilités pour un agent, puis les filtre pour enlever
     * les faux positifs connus grâce à notre filtre OVAL.
     *
     * @param agentId L'ID de l'agent.
     * @return Une liste de vulnérabilités nettoyée.
     */
    public List<Vulnerability> getVulnerabilitiesForAgent(String agentId) {
        System.out.println("=== Recherche des vulnérabilités pour l'agent: " + agentId + " ===");
        
        try {
            // La requête OpenSearch demande tous les champs dont nous avons besoin
            String query = String.format("""
                {
                  "size": 10000,
                  "_source": [
                      "vulnerability.id",
                      "vulnerability.severity",
                      "vulnerability.description",
                      "vulnerability.published_at",
                      "vulnerability.score.base",
                      "vulnerability.reference",
                      "package.name",
                      "package.version"
                  ],
                  "query": {
                    "bool": {
                      "must": [
                        { "term": { "agent.id": "%s" } }
                      ],
                      "filter": [
                        { "terms": { "vulnerability.severity": ["Critical", "High"] } }
                      ]
                    }
                  }
                }
            """, agentId);

            JsonNode response = indexerWebClient.post()
                    .uri("/wazuh-states-vulnerabilities-x3rsi/_search")
                    .header("Content-Type", "application/json")
                    .bodyValue(query)
                    .retrieve()
                    .bodyToMono(JsonNode.class)
                    .block();

            // Étape 1 : On parse la réponse brute de l'API
            List<Vulnerability> rawVulnerabilities = parseVulnerabilityResponse(response);
            
            if (!rawVulnerabilities.isEmpty()) {
                System.out.println("--- [DIAGNOSTIC OVAL] Noms de paquets reçus de Wazuh (échantillon) ---");
                rawVulnerabilities.stream()
                    .map(Vulnerability::getName)
                    .distinct()
                    .limit(15) // On ne log que les 15 premiers noms uniques pour ne pas polluer
                    .forEach(name -> System.out.println("WAZUH_PKG_NAME: " + name));
                System.out.println("------------------------------------------------------------------");
            }
            // --- FIN DE L'AJOUT ---
            
            // Étape 2 : On applique notre filtre intelligent pour retirer les faux positifs
            List<Vulnerability> filteredVulnerabilities = rawVulnerabilities.stream()
                .filter(vuln -> !ovalFilter.isFalsePositive(vuln))
                .collect(Collectors.toList());
                
            System.out.println(rawVulnerabilities.size() + " vulnérabilités brutes trouvées, " + 
                               filteredVulnerabilities.size() + " restantes après filtrage OVAL.");
                               
            return filteredVulnerabilities;
            
        } catch (Exception e) {
            System.err.println("Erreur lors de la recherche des vulnérabilités: " + e.getMessage());
            e.printStackTrace();
            return Collections.emptyList();
        }
    }
    
    /**
     * Parse la réponse JSON d'OpenSearch pour la transformer en une liste d'objets Vulnerability.
     */
    private List<Vulnerability> parseVulnerabilityResponse(JsonNode response) {
        List<Vulnerability> vulnerabilities = new ArrayList<>();
        if (response == null || !response.has("hits") || !response.path("hits").has("hits")) {
            return vulnerabilities;
        }
        
        for (JsonNode hit : response.path("hits").path("hits")) {
            JsonNode source = hit.path("_source");
            if (source.isMissingNode()) continue;

            JsonNode vulnNode = source.path("vulnerability");
            JsonNode packageNode = source.path("package");
            
            Vulnerability vuln = new Vulnerability();
            
            // On utilise les bons noms de champs identifiés grâce au JSON brut
            vuln.setCve(vulnNode.path("id").asText("N/A")); 
            vuln.setTitle(vulnNode.path("description").asText("Aucune description")); 
            vuln.setPublishedDate(vulnNode.path("published_at").asText(null));
            vuln.setSeverity(vulnNode.path("severity").asText("Inconnue"));
            vuln.setName(packageNode.path("name").asText("N/A"));
            vuln.setVersion(packageNode.path("version").asText("N/A"));
            vuln.setScore(vulnNode.path("score").path("base").asDouble(0.0));

            String references = vulnNode.path("reference").asText("");
            if (!references.isEmpty()) {
                vuln.setReference(references.split(",")[0].trim());
            } else {
                vuln.setReference(null);
            }
            
            vulnerabilities.add(vuln);
        }
        return vulnerabilities;
    }

    /**
     * WebClient pour l'API Wazuh en mode DÉVELOPPEMENT.
     * Désactive complètement la vérification SSL.
     */
    @Bean("wazuhApiWebClient")
    @Profile({"dev", "preprod"})
    public WebClient insecureWazuhApiWebClient(@Value("${wazuh.api.url}") String wazuhApiUrl) {
        System.out.println("--- PROFIL DEV : Création d'un WebClient NON SÉCURISÉ pour l'API Wazuh ---");
        try {
            SslContext sslContext = SslContextBuilder.forClient()
                    .trustManager(InsecureTrustManagerFactory.INSTANCE)
                    .build();
            HttpClient httpClient = HttpClient.create().secure(t -> t.sslContext(sslContext));

            // Configuration des stratégies d'échange avec limite augmentée
            ExchangeStrategies strategies = ExchangeStrategies.builder()
                    .codecs(clientCodecConfigurer -> {
                        clientCodecConfigurer.defaultCodecs().maxInMemorySize(MAX_IN_MEMORY_SIZE);
                    })
                    .build();

            return WebClient.builder()
                    .baseUrl(wazuhApiUrl)
                    .clientConnector(new ReactorClientHttpConnector(httpClient))
                    .exchangeStrategies(strategies)
                    .build();
        } catch (Exception e) {
            throw new RuntimeException("Échec de la configuration du WebClient non sécurisé pour l'API Wazuh.", e);
        }

import org.bouncycastle.asn1.ASN1ObjectIdentifier;
import org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.bouncycastle.tsp.TSPAlgorithms; // <-- L'IMPORT CORRECTimport org.bouncycastle.jce.provider.BouncyCastleProvider;
import org.bouncycastle.tsp.TimeStampRequest;
import org.bouncycastle.tsp.TimeStampRequestGenerator;
import org.bouncycastle.tsp.TimeStampResponse;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import java.io.ByteArrayOutputStream;
import java.io.IOException;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.security.MessageDigest;
import java.security.Security;

@Service
public class TimestampingService {

    private final String tsaUrl;

    public TimestampingService(@Value("${services.tsa.url}") String tsaUrl) {
        this.tsaUrl = tsaUrl;
        // Ajoute Bouncy Castle comme fournisseur de sécurité si ce n'est pas déjà fait
        Security.addProvider(new BouncyCastleProvider());
    }

    /**
     * Crée une réponse d'horodatage (TSR) pour les données fournies.
     * @param data Les données à horodater (ex: les bytes d'un PDF).
     * @return Un tableau de bytes représentant la réponse d'horodatage (.tsr).
     * @throws Exception Si une erreur survient pendant le processus.
     */
    public byte[] createTimestampResponse(byte[] data) throws Exception {
        // 1. Calculer le hash SHA-256 des données
        MessageDigest digest = MessageDigest.getInstance("SHA-256");
        byte[] hash = digest.digest(data);
        System.out.println("Hash SHA-256 du document calculé.");

        // 2. Générer une requête d'horodatage (TSQ) avec ce hash
        TimeStampRequestGenerator reqGen = new TimeStampRequestGenerator();
        reqGen.setCertReq(true); // On demande à la TSA d'inclure son certificat dans la réponse
        ASN1ObjectIdentifier oid = TSPAlgorithms.SHA256;
        TimeStampRequest request = reqGen.generate(oid, hash);
        byte[] requestBytes = request.getEncoded();
        System.out.println("Requête d'horodatage (TSQ) générée.");

        // 3. Envoyer la requête à l'Autorité d'Horodatage (TSA) via HTTP POST
        System.out.println("Envoi de la requête à la TSA : " + tsaUrl);
        URL url = new URL(tsaUrl);
        HttpURLConnection connection = (HttpURLConnection) url.openConnection();
        connection.setRequestMethod("POST");
        connection.setDoOutput(true);
        connection.setRequestProperty("Content-Type", "application/timestamp-query");
        
        try (OutputStream out = connection.getOutputStream()) {
            out.write(requestBytes);
        }

        if (connection.getResponseCode() != HttpURLConnection.HTTP_OK) {
            throw new IOException("Réponse inattendue de la TSA : " + connection.getResponseCode() + " " + connection.getResponseMessage());
        }

        // 4. Lire la réponse d'horodatage (TSR)
        try (InputStream in = connection.getInputStream(); ByteArrayOutputStream baos = new ByteArrayOutputStream()) {
            byte[] buffer = new byte[1024];
            int bytesRead;
            while ((bytesRead = in.read(buffer)) != -1) {
                baos.write(buffer, 0, bytesRead);
            }
            byte[] responseBytes = baos.toByteArray();
            System.out.println("Réponse d'horodatage (TSR) reçue.");

            // 5. Valider la réponse (facultatif mais recommandé)
            TimeStampResponse response = new TimeStampResponse(responseBytes);
            response.validate(request);
            System.out.println("Réponse d'horodatage validée avec succès.");

            return response.getEncoded();
        } finally {
            connection.disconnect();
        }
    }

import xns.x3rsi.repository.CveReferenceRepository;
import xns.x3rsi.repository.PatchedPackageRepository;
import xns.x3rsi.wazuh.dto.Vulnerability;

import javax.xml.stream.XMLInputFactory;
import javax.xml.stream.XMLStreamConstants;
import javax.xml.stream.XMLStreamReader;
import java.io.BufferedInputStream;
import java.io.InputStream;
import java.util.ArrayList;
import java.util.Collections;
import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.concurrent.ConcurrentHashMap;

@Service
public class OvalVulnerabilityFilter {

    private static final Logger log = LoggerFactory.getLogger(OvalVulnerabilityFilter.class);

    // --- CONFIGURATION ---
    private static final String OVAL_FILE_PATH = "classpath:feeds/com.ubuntu.jammy.cve.oval.xml";
    private static final int BATCH_SIZE = 500; // Pour l'insertion en BDD
    private static final int BUFFER_SIZE = 256 * 1024; // Buffer de lecture du XML

    @Value("${oval.parsing.enabled:true}")
    private boolean parsingEnabled;

    // --- DÉPENDANCES ---
    private final ResourceLoader resourceLoader;
    private final CveReferenceRepository cveRepository;
    private final PatchedPackageRepository packageRepository;
    private final TransactionTemplate transactionTemplate;

    // --- LE CACHE EN MÉMOIRE OPTIMISÉ (NOTRE OBJECTIF FINAL) ---
    // Structure : Map<NomDuPaquetBinaire, Set<CVE_ID>>
    // C'est notre "bureau du bibliothécaire", optimisé pour la vitesse et la mémoire.
    private volatile Map<String, Set<String>> packageToCvesCache;

    public OvalVulnerabilityFilter(ResourceLoader resourceLoader,
                                   CveReferenceRepository cveRepository,
                                   PatchedPackageRepository packageRepository,
                                   PlatformTransactionManager transactionManager) {
        this.resourceLoader = resourceLoader;
        this.cveRepository = cveRepository;
        this.packageRepository = packageRepository;
        this.transactionTemplate = new TransactionTemplate(transactionManager);
    }

    /**
     * Méthode principale exécutée au démarrage de l'application.
     * C'est le point d'entrée de notre nouvelle logique.
     */
    @PostConstruct
    public void initialize() {
        if (!parsingEnabled) {
            log.info("Filtrage OVAL désactivé par configuration.");
            return;
        }
        log.info("=== Initialisation du filtre de vulnérabilités OVAL ===");
        
        // On vérifie si la BDD est déjà peuplée.
        if (packageRepository.count() > 0) {
            log.info("La base de données OVAL est déjà peuplée. Construction du cache depuis la BDD...");
            buildCacheFromDatabase();
        } else {
            log.info("La base de données OVAL est vide. Lancement de l'ingestion depuis le fichier XML...");
            ingestOvalData();
            // Une fois l'ingestion terminée, on construit le cache depuis la BDD fraîchement remplie.
            buildCacheFromDatabase();
        }
    }

    /**
     * Nouvelle méthode d'ingestion. Elle parse le XML et peuple la base de données.
     * C'est le "bibliothécaire qui lit le livre ancien une seule fois".
     */
    public void ingestOvalData() {
        long startTime = System.currentTimeMillis();
        try {
            // Étape 1: Pré-traitement. On parcourt le XML pour construire les cartes de relations.
            log.info("[Phase 1/3] Analyse des relations dans le fichier OVAL...");
            Map<String, String> testIdToObjectRef = new HashMap<>();
            Map<String, String> objectIdToVarRef = new HashMap<>();
            Map<String, List<String>> varIdToPackages = new HashMap<>();
            preProcessOvalFile(testIdToObjectRef, objectIdToVarRef, varIdToPackages);
            log.info("✓ Relations analysées : {} tests, {} objets, {} variables de paquets.", 
                testIdToObjectRef.size(), objectIdToVarRef.size(), varIdToPackages.size());

            // Étape 2: Passe principale. On parcourt à nouveau pour lier les CVEs aux paquets.
            log.info("[Phase 2/3] Association des CVEs aux paquets binaires...");
            Map<String, Set<String>> cveToPackages = processDefinitions(testIdToObjectRef, objectIdToVarRef, varIdToPackages);
            log.info("✓ {} CVEs associées à leurs paquets corrigés.", cveToPackages.size());

            // Étape 3: Sauvegarde en base de données.
            log.info("[Phase 3/3] Sauvegarde des relations dans la base de données...");
            saveToDatabase(cveToPackages);
            log.info("✓ Sauvegarde terminée.");

        } catch (Exception e) {
            log.error("ERREUR CRITIQUE lors de l'ingestion des données OVAL. Le filtrage sera inopérant.", e);
        }
        long duration = System.currentTimeMillis() - startTime;
        log.info("Ingestion OVAL terminée en {} secondes.", duration / 1000);
    }

    /**
     * Construit le cache en mémoire en lisant la base de données.
     * Rapide et efficace.
     */
    private void buildCacheFromDatabase() {
        long startTime = System.currentTimeMillis();
        Map<String, Set<String>> cache = new ConcurrentHashMap<>();
        
        // On récupère tous les paquets avec leurs CVEs corrigées en une seule fois.
        List<PatchedPackage> allPackages = packageRepository.findAllWithCves();
        
        for (PatchedPackage pkg : allPackages) {
            Set<String> cveIds = new HashSet<>();
            for (CveReference cve : pkg.getFixedCves()) {
                cveIds.add(cve.getId());
            }
            cache.put(pkg.getName(), cveIds);
        }
        
        this.packageToCvesCache = cache;
        long duration = System.currentTimeMillis() - startTime;
        log.info("✓ Cache OVAL construit depuis la BDD en {} ms. {} paquets chargés.", duration, cache.size());
    }

    // --- MÉTHODES DE PARSING (LE CŒUR DE LA NOUVELLE LOGIQUE) ---

    private void preProcessOvalFile(Map<String, String> testIdToObjectRef, Map<String, String> objectIdToVarRef, Map<String, List<String>> varIdToPackages) throws Exception {
        try (InputStream is = new BufferedInputStream(resourceLoader.getResource(OVAL_FILE_PATH).getInputStream(), BUFFER_SIZE)) {
            XMLStreamReader reader = createOptimizedReader(is);
            while (reader.hasNext()) {
                if (reader.getEventType() == XMLStreamConstants.START_ELEMENT) {
                    String localName = reader.getLocalName();
                    if ("dpkginfo_test".equals(localName)) {
                        String testId = reader.getAttributeValue(null, "id");
                        reader.nextTag();
                        String objectRef = reader.getAttributeValue(null, "object_ref");
                        if (testId != null && objectRef != null) {
                            testIdToObjectRef.put(testId, objectRef);
                        }
                    } else if ("dpkginfo_object".equals(localName)) {
                        String objectId = reader.getAttributeValue(null, "id");
                        reader.nextTag();
                        String varRef = reader.getAttributeValue(null, "var_ref");
                        if (objectId != null && varRef != null) {
                            objectIdToVarRef.put(objectId, varRef);
                        }
                    } else if ("constant_variable".equals(localName)) {
                        String varId = reader.getAttributeValue(null, "id");
                        List<String> packages = new ArrayList<>();
                        while (reader.hasNext()) {
                            int event = reader.next();
                            if (event == XMLStreamConstants.START_ELEMENT && "value".equals(reader.getLocalName())) {
                                packages.add(reader.getElementText());
                            } else if (event == XMLStreamConstants.END_ELEMENT && "constant_variable".equals(reader.getLocalName())) {
                                break;
                            }
                        }
                        if (varId != null && !packages.isEmpty()) {
                            varIdToPackages.put(varId, packages);
                        }
                    }
                }
                reader.next();
            }
        }
    }
    
    private Map<String, Set<String>> processDefinitions(Map<String, String> testIdToObjectRef, Map<String, String> objectIdToVarRef, Map<String, List<String>> varIdToPackages) throws Exception {
        Map<String, Set<String>> cveToPackages = new HashMap<>();
        try (InputStream is = new BufferedInputStream(resourceLoader.getResource(OVAL_FILE_PATH).getInputStream(), BUFFER_SIZE)) {
            XMLStreamReader reader = createOptimizedReader(is);
            String currentCVE = null;
            while (reader.hasNext()) {
                int event = reader.next();
                if (event == XMLStreamConstants.START_ELEMENT) {
                    String localName = reader.getLocalName();
                    if ("definition".equals(localName) && "vulnerability".equals(reader.getAttributeValue(null, "class"))) {
                        // On est au début d'une définition de vulnérabilité
                    } else if ("reference".equals(localName) && "CVE".equals(reader.getAttributeValue(null, "source"))) {
                        currentCVE = reader.getAttributeValue(null, "ref_id");
                    } else if ("criterion".equals(localName)) {
                        String testRef = reader.getAttributeValue(null, "test_ref");
                        if (currentCVE != null && testRef != null) {
                            String objectRef = testIdToObjectRef.get(testRef);
                            if (objectRef != null) {
                                String varRef = objectIdToVarRef.get(objectRef);
                                if (varRef != null) {
                                    List<String> packages = varIdToPackages.get(varRef);
                                    if (packages != null) {
                                        cveToPackages.computeIfAbsent(currentCVE, k -> new HashSet<>()).addAll(packages);
                                    }
                                }
                            }
                        }
                    }
                } else if (event == XMLStreamConstants.END_ELEMENT && "definition".equals(reader.getLocalName())) {
                    currentCVE = null; // Reset for the next definition
                }
            }
        }
        return cveToPackages;
    }

    private void saveToDatabase(Map<String, Set<String>> cveToPackages) {
        // On inverse la map pour avoir package -> CVEs
        Map<String, Set<String>> packageToCves = new HashMap<>();
        cveToPackages.forEach((cve, packages) -> {
            for (String pkgName : packages) {
                packageToCves.computeIfAbsent(pkgName, k -> new HashSet<>()).add(cve);
            }
        });

        // On utilise un TransactionTemplate pour gérer les transactions manuellement par lots
        transactionTemplate.execute(status -> {
            List<PatchedPackage> batch = new ArrayList<>();
            for (Map.Entry<String, Set<String>> entry : packageToCves.entrySet()) {
                String pkgName = entry.getKey();
                Set<String> cveIds = entry.getValue();

                PatchedPackage pkg = new PatchedPackage(pkgName);
                
                // On récupère ou crée les CveReference
                List<CveReference> cveRefs = cveRepository.findAllById(cveIds);
                Map<String, CveReference> existingCves = new HashMap<>();
                cveRefs.forEach(c -> existingCves.put(c.getId(), c));
                
                for (String cveId : cveIds) {
                    CveReference cveRef = existingCves.computeIfAbsent(cveId, CveReference::new);
                    pkg.getFixedCves().add(cveRef);
                }
                batch.add(pkg);

                if (batch.size() >= BATCH_SIZE) {
                    log.info("Sauvegarde d'un lot de {} paquets...", batch.size());
                    packageRepository.saveAllAndFlush(batch);
                    batch.clear();
                }
            }
            if (!batch.isEmpty()) {
                log.info("Sauvegarde du dernier lot de {} paquets...", batch.size());
                packageRepository.saveAllAndFlush(batch);
            }
            return null;
        });
    }

    // --- LA MÉTHODE DE FILTRAGE PUBLIQUE (SIMPLE ET RAPIDE) ---

    /**
     * Vérifie si une vulnérabilité est un faux positif en consultant le cache en mémoire.
     */
    public boolean isFalsePositive(Vulnerability vulnerability) {
        if (packageToCvesCache == null || vulnerability == null || vulnerability.getCve() == null || vulnerability.getName() == null) {
            return false;
        }

        // Étape 1: Normaliser le nom du paquet venant de Wazuh (ex: enlever ":amd64")
        String normalizedPkgName = vulnerability.getName().split(":")[0];

        // Étape 2: Consulter le cache
        Set<String> fixedInCves = packageToCvesCache.get(normalizedPkgName);
        if (fixedInCves == null) {
            return false; // Ce paquet n'est listé dans aucun correctif OVAL que nous connaissons.
        }

        // Étape 3: Vérifier si le CVE de la vulnérabilité est dans la liste des correctifs pour ce paquet
        boolean isFP = fixedInCves.contains(vulnerability.getCve());
        
        if (isFP) {
            // Log uniquement si on trouve un faux positif, pour le débogage
            log.debug("Faux positif détecté et filtré -> Paquet: {}, CVE: {}", normalizedPkgName, vulnerability.getCve());
        }
        
        return isFP;
    }

    // --- UTILITAIRES ---

    private XMLStreamReader createOptimizedReader(InputStream is) throws Exception {
        XMLInputFactory factory = XMLInputFactory.newInstance();
        factory.setProperty(XMLInputFactory.IS_SUPPORTING_EXTERNAL_ENTITIES, false);
        factory.setProperty(XMLInputFactory.SUPPORT_DTD, false);
        return factory.createXMLStreamReader(is);
    }

import org.springframework.beans.factory.annotation.Qualifier;

@Service
public class WazuhAlertsService {

    private final WebClient indexerWebClient;
    private final CustomerGroupRepository groupRepository;
    private final SimpMessagingTemplate messagingTemplate;
    private Instant lastCheckTimestamp;

    public WazuhAlertsService(@Qualifier("indexerWebClient") WebClient indexerWebClient,
                              CustomerGroupRepository groupRepository,
                              SimpMessagingTemplate messagingTemplate) {
        this.indexerWebClient = indexerWebClient;
        this.groupRepository = groupRepository;
        this.messagingTemplate = messagingTemplate;
        this.lastCheckTimestamp = Instant.now().minus(1, ChronoUnit.MINUTES);
    }

    @Scheduled(fixedRate = 1000)
    public void fetchAndPushNewAlerts() {
        // System.out.println("Vérification des nouvelles alertes Wazuh..."); // Optionnel: décommenter pour le debug
        Instant now = Instant.now();
        try {
            String query = String.format("""
                { "size": 500, "query": { "range": { "@timestamp": { "gt": "%s", "lte": "%s" } } }, "sort": [ { "@timestamp": "asc" } ] }
            """, lastCheckTimestamp.toString(), now.toString());

            JsonNode response = indexerWebClient.post()
                .uri("/wazuh-alerts-*/_search")
                .header("Content-Type", "application/json")
                .bodyValue(query)
                .retrieve()
                .bodyToMono(JsonNode.class)
                .block();

            List<WazuhAlert> newAlerts = parseAlertsResponse(response);

            if (!newAlerts.isEmpty()) {
                System.out.println(newAlerts.size() + " nouvelle(s) alerte(s) trouvee(s). Envoi en cours...");
                Map<String, Long> agentToGroupIdMap = buildAgentToGroupMap();
                for (WazuhAlert alert : newAlerts) {
                    Long groupId = agentToGroupIdMap.get(alert.getAgentId());
                    if (groupId != null) {
                        String destination = "/topic/alerts/group/" + groupId;
                        messagingTemplate.convertAndSend(destination, alert);
                        messagingTemplate.convertAndSend("/topic/alerts/all", alert);
                    }
                }
            }
        } catch (Exception e) {
            System.err.println("Erreur lors du cycle de récupération des alertes: " + e.getMessage());
        }
        this.lastCheckTimestamp = now;
    }

    @Transactional(readOnly = true)
    public Map<String, Long> buildAgentToGroupMap() {
        return groupRepository.findAll().stream()
            .flatMap(group -> group.getAgents().stream()
                .map(agent -> Map.entry(agent.getId(), group.getId())))
            .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue, (id1, id2) -> id1));
    }

    @Transactional(readOnly = true)
    public List<WazuhAlert> getHistoricAlerts(String userRole, Long groupId) {
        Instant now = Instant.now();
        Instant yesterday = now.minus(24, ChronoUnit.HOURS);
        String agentFilterClause = "";
        
        if ("ROLE_ADMIN".equals(userRole)) {
            if (groupId == null) return Collections.emptyList();
            CustomerGroup group = groupRepository.findById(groupId)
                    .orElseThrow(() -> new RuntimeException("Groupe non trouvé pour l'historique: " + groupId));
            String agentIds = group.getAgents().stream()
                                .map(agent -> "\"" + agent.getId() + "\"")
                                .collect(Collectors.joining(", "));
            if (!agentIds.isEmpty()) {
                agentFilterClause = String.format(",{ \"terms\": { \"agent.id\": [%s] } }", agentIds);
            } else {
                return Collections.emptyList();
            }
        }

        String query = String.format(
            "{ \"size\": 500, \"query\": { \"bool\": { \"must\": [ { \"range\": { \"@timestamp\": { \"gte\": \"%s\", \"lte\": \"%s\" } } }%s ] } }, \"sort\": [ { \"@timestamp\": \"desc\" } ] }",
            yesterday.toString(), now.toString(), agentFilterClause
        );

        JsonNode response = indexerWebClient.post()
                .uri("/wazuh-alerts-*/_search")
                .header("Content-Type", "application/json")
                .bodyValue(query)
                .retrieve()
                .bodyToMono(JsonNode.class)
                .block();

        return parseAlertsResponse(response);
    }

    private List<WazuhAlert> parseAlertsResponse(JsonNode response) {
        List<WazuhAlert> alerts = new ArrayList<>();
        if (response == null || !response.has("hits")) return alerts;
        for (JsonNode hit : response.path("hits").path("hits")) {
            JsonNode source = hit.path("_source");
            WazuhAlert alert = new WazuhAlert();
            alert.setTimestamp(source.path("@timestamp").asText());
            alert.setAgentName(source.path("agent").path("name").asText("N/A"));
            alert.setAgentId(source.path("agent").path("id").asText("N/A"));
            alert.setRuleId(source.path("rule").path("id").asText("N/A"));
            alert.setRuleDescription(source.path("rule").path("description").asText("N/A"));
            alert.setRuleLevel(source.path("rule").path("level").asInt(0));
            alerts.add(alert);
        }
        return alerts;
    }

import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import xns.x3rsi.model.Agent;
import xns.x3rsi.model.CustomerGroup;
import xns.x3rsi.model.User;
import xns.x3rsi.model.enums.Role;
import xns.x3rsi.repository.AgentRepository;
import xns.x3rsi.repository.UserRepository;

@Service
public class UserService {

    private final UserRepository userRepository;
    private final AgentRepository agentRepository;
    private final PasswordEncoder passwordEncoder;
    private final CustomerGroupService customerGroupService;

    public UserService(UserRepository userRepository,
                         AgentRepository agentRepository,
                         PasswordEncoder passwordEncoder,
                         CustomerGroupService customerGroupService) {
        this.userRepository = userRepository;
        this.agentRepository = agentRepository;
        this.passwordEncoder = passwordEncoder;
        this.customerGroupService = customerGroupService;
    }

    @Transactional
    public User createUser(String username, String email, String plainTextPassword, Role role, Long groupId) {
        userRepository.findByUsername(username).ifPresent(u -> {
            throw new IllegalStateException("Le nom d'utilisateur existe déjà: " + username);
        });
        userRepository.findByEmail(email).ifPresent(u -> {
            throw new IllegalStateException("L'adresse e-mail est déjà utilisée: " + email);
        });

        Role userRole = (role == null) ? Role.ROLE_USER : role;
        String hashedPassword = passwordEncoder.encode(plainTextPassword);
        User newUser = new User(username, email, hashedPassword, userRole);

        if (groupId != null) {
            CustomerGroup group = customerGroupService.findById(groupId);
            newUser.setCustomerGroup(group);
        } else if (userRole == Role.ROLE_USER || userRole == Role.ROLE_ADMIN) {
            throw new IllegalStateException("Un groupe est obligatoire pour les rôles USER et ADMIN.");
        }

        return userRepository.save(newUser);
    }

    @Transactional
    public User assignUserToGroup(String username, Long groupId) {
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new RuntimeException("Utilisateur non trouvé: " + username));
        
        CustomerGroup group = customerGroupService.findById(groupId);
        user.setCustomerGroup(group);

        // Si l'utilisateur est un admin, on lui donne accès à tous les agents déjà présents dans le groupe
        if (user.getRole() == Role.ROLE_ADMIN) {
            user.getAgents().addAll(group.getAgents());
        }

        return userRepository.save(user);
    }

    @Transactional
    public void assignAgentToUser(String username, String agentId, String agentName) {
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new RuntimeException("Utilisateur non trouvé: " + username));

        if (user.getRole() == Role.ROLE_USER && !user.getAgents().isEmpty()) {
            throw new IllegalStateException("Un utilisateur avec le rôle ROLE_USER ne peut être associé qu'à un seul agent.");
        }

        Agent agent = agentRepository.findById(agentId).orElseGet(() -> {
            Agent newAgent = new Agent(agentId, agentName);
            return agentRepository.save(newAgent);
        });

        user.getAgents().add(agent);
        userRepository.save(user);
    }

    public boolean checkPassword(String username, String plainTextPassword) {
        return userRepository.findByUsername(username)
                .map(user -> passwordEncoder.matches(plainTextPassword, user.getPassword()))
                .orElse(false);
    }

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.SignatureAlgorithm;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.stereotype.Service;
import xns.x3rsi.model.User;

import javax.crypto.SecretKey;
import java.util.Date;
import java.util.HashMap;
import java.util.Map;
import java.util.function.Function;

@Service
public class JwtService {

    @Value("${jwt.secret:myVerySecretKeyForX3RSIThatShouldBeInConfigFile}")
    private String secretKey;

    @Value("${jwt.expiration:86400000}")
    private long jwtExpiration;

    public String generateToken(User user) {
        Map<String, Object> extraClaims = new HashMap<>();
        extraClaims.put("role", user.getRole().name());
        if (user.getCustomerGroup() != null) {
            extraClaims.put("groupId", user.getCustomerGroup().getId());
        }
        return buildToken(extraClaims, String.valueOf(user.getId()), jwtExpiration);
    }

    public Long extractUserId(String token) {
        String subject = extractClaim(token, Claims::getSubject);
        return Long.parseLong(subject);
    }

    public boolean isTokenValid(String token, UserDetails userDetails) {
        final Long userId = extractUserId(token);
        // Le "username" de UserDetails est maintenant notre ID
        return (userId.toString().equals(userDetails.getUsername()) && !isTokenExpired(token));
    }
    
    // --- Méthodes utilitaires (inchangées) ---

    public <T> T extractClaim(String token, Function<Claims, T> claimsResolver) {
        final Claims claims = extractAllClaims(token);
        return claimsResolver.apply(claims);
    }

    private boolean isTokenExpired(String token) {
        return extractExpiration(token).before(new Date());
    }

    private Date extractExpiration(String token) {
        return extractClaim(token, Claims::getExpiration);
    }

    private String buildToken(Map<String, Object> extraClaims, String subject, long expiration) {
        return Jwts.builder()
                .setClaims(extraClaims)
                .setSubject(subject)
                .setIssuedAt(new Date(System.currentTimeMillis()))
                .setExpiration(new Date(System.currentTimeMillis() + expiration))
                .signWith(getSignInKey(), SignatureAlgorithm.HS256)
                .compact();
    }

    private Claims extractAllClaims(String token) {
        return Jwts.parserBuilder()
                .setSigningKey(getSignInKey())
                .build()
                .parseClaimsJws(token)
                .getBody();
    }

    private SecretKey getSignInKey() {
        return Keys.hmacShaKeyFor(secretKey.getBytes());
    }

@Service
public class CustomerGroupService {

    private final CustomerGroupRepository groupRepository;
    private final AgentRepository agentRepository;

    public CustomerGroupService(CustomerGroupRepository groupRepository, AgentRepository agentRepository, UserRepository userRepository) {
        this.groupRepository = groupRepository;
        this.agentRepository = agentRepository;
        this.userRepository = userRepository;
    }

    @Transactional
    public CustomerGroup createGroup(String name) {
        // La condition est maintenant valide
        groupRepository.findByName(name).ifPresent(g -> {
            throw new IllegalStateException("Un groupe avec le nom '" + name + "' existe déjà.");

    public byte[] generateReportZip(String agentId, Map<String, String> qcmAnswers) throws Exception {
        Path tempDir = Files.createTempDirectory("report-" + agentId + "-");
        try {
            // --- 1. RÉCUPÉRATION ET SAUVEGARDE DES DONNÉES BRUTES ---
            List<Vulnerability> vulnerabilities = vulnerabilityService.getVulnerabilitiesForAgent(agentId);
            objectMapper.writerWithDefaultPrettyPrinter().writeValue(tempDir.resolve("vuln_result.json").toFile(), vulnerabilities);

            List<ScaPolicySummary> scaSummaries = scaService.getScaSummaryForAgent(agentId);
            List<ScaCheck> scaChecks = Collections.emptyList();
            ScaPolicySummary ourPolicy = scaSummaries.stream()
                .filter(p -> p.getName() != null && p.getName().toUpperCase().contains("XNS"))
                .findFirst().orElse(null);

            if (ourPolicy != null) {
                scaChecks = scaService.getAllScaChecks(ourPolicy.getPolicyId(), agentId);
            }
            objectMapper.writerWithDefaultPrettyPrinter().writeValue(tempDir.resolve("sca_result.json").toFile(), scaChecks);

            // --- 2. PRÉPARATION DU CONTEXTE POUR LE TEMPLATE ---
            Map<String, Object> templateVariables = prepareTemplateVariables(agentId, vulnerabilities, scaChecks, ourPolicy, qcmAnswers);

            // --- 3. GÉNÉRATION DU PDF ---
            String htmlContent = templateService.processTemplate("report-template.html", templateVariables);
            Document doc = Jsoup.parse(htmlContent);
            doc.outputSettings().syntax(Document.OutputSettings.Syntax.xml);
            String xhtmlContent = doc.html();

            Path pdfPath = tempDir.resolve("rapport_rgpd_xns.pdf");
            byte[] pdfBytes;
            try (ByteArrayOutputStream os = new ByteArrayOutputStream()) {
                ITextRenderer renderer = new ITextRenderer();
                
                // Correction du chemin de base pour qu'il fonctionne dans le JAR
                String baseUrl = getClass().getResource("/templates/").toExternalForm();
                
                renderer.setDocumentFromString(xhtmlContent, baseUrl);
                renderer.layout();
                renderer.createPDF(os);
                pdfBytes = os.toByteArray();
                Files.write(pdfPath, pdfBytes);
            }

            // --- 4. HORODATAGE DU PDF ---
            MessageDigest digest = MessageDigest.getInstance("SHA-256");
            byte[] pdfHash = digest.digest(pdfBytes);
            String sha256Hex = toHexString(pdfHash);
            Path sha256Path = tempDir.resolve("rapport_rgpd_xns.sha256");
            Files.writeString(sha256Path, sha256Hex);

            try {
                byte[] tsrBytes = timestampingService.createTimestampResponse(pdfBytes);
                Path tsrPath = tempDir.resolve("rapport_rgpd_xns.tsr");
                Files.write(tsrPath, tsrBytes);
            } catch (Exception e) {
                System.err.println("ERREUR d'horodatage: " + e.getMessage());
            }

            // --- 5. ASSEMBLAGE DU ZIP ---
            return createZipArchive(tempDir);

        } finally {
            FileUtils.deleteDirectory(tempDir.toFile());
        }
    }

    private Map<String, Object> prepareTemplateVariables(String agentId, List<Vulnerability> vulnerabilities, List<ScaCheck> scaChecks, ScaPolicySummary ourPolicy, Map<String, String> qcmAnswers) {
        long criticalVulnCount = vulnerabilities.stream().filter(v -> "Critical".equalsIgnoreCase(v.getSeverity())).count();
        long highVulnCount = vulnerabilities.stream().filter(v -> "High".equalsIgnoreCase(v.getSeverity())).count();
        boolean noCriticalVulnerabilities = (criticalVulnCount == 0);
        boolean scaScoreIsAcceptable = false;
        double scaScorePercentage = 0.0;
        if (ourPolicy != null && ourPolicy.getTotalChecks() > 0) {
            scaScorePercentage = ((double) ourPolicy.getPassCount() / ourPolicy.getTotalChecks()) * 100;
            scaScoreIsAcceptable = scaScorePercentage >= 90.0;
        }
        boolean isConforme = noCriticalVulnerabilities && scaScoreIsAcceptable;
        String finalConclusionStatus = isConforme ? "OUI, VOUS ÊTES CONFORME" : "NON, VOUS ÊTES NON CONFORME";
        String finalRecommendation = isConforme
            ? "Le système respecte les exigences techniques analysées. Maintenez ce niveau de vigilance."
            : "Des actions correctives sont nécessaires sur le plan technique pour atteindre la conformité.";
        Optional<Agent> agentOptional = agentRepository.findById(agentId);
        String clientName = agentOptional.flatMap(agent -> agent.getUsers().stream().findFirst())
            .map(user -> user.getCustomerGroup().getName()).orElse("Client non spécifié");
        
        Map<String, Object> templateVariables = new HashMap<>();
        templateVariables.put("criticalVulnCount", criticalVulnCount);
        templateVariables.put("highVulnCount", highVulnCount);
        templateVariables.put("finalConclusionStatus", finalConclusionStatus);
        templateVariables.put("finalRecommendation", finalRecommendation);
        templateVariables.put("checkNoCriticalVulnerabilities", noCriticalVulnerabilities);
        templateVariables.put("checkScaScore", scaScoreIsAcceptable);
        templateVariables.put("scaScorePercentage", String.format(Locale.US, "%.1f%%", scaScorePercentage));
        templateVariables.put("generationDate", LocalDate.now().format(DateTimeFormatter.ofPattern("dd MMMM yyyy", Locale.FRENCH)));
        templateVariables.put("clientName", clientName);
        templateVariables.put("machineName", "Agent ID: " + agentId);
        templateVariables.put("vulnerabilities", vulnerabilities);
        List<ScaCheckViewModel> scaCheckViewModels = scaChecks.stream().map(ScaCheckViewModel::new).collect(Collectors.toList());
        templateVariables.put("scaChecks", scaCheckViewModels);
        templateVariables.put("qcmAnswers", qcmAnswers);
        if (ourPolicy != null) {
            templateVariables.put("totalScaChecks", ourPolicy.getTotalChecks());
            templateVariables.put("passedScaChecks", ourPolicy.getPassCount());
            templateVariables.put("failedScaChecks", ourPolicy.getFailCount());
            templateVariables.put("scaScore", ourPolicy.getPassCount());
        } else {
            templateVariables.put("totalScaChecks", 0);
            templateVariables.put("passedScaChecks", 0);
            templateVariables.put("failedScaChecks", 0);
            templateVariables.put("scaScore", 0);
        }
        return templateVariables;
    }

    private byte[] createZipArchive(Path sourceDir) throws IOException {
        ByteArrayOutputStream baos = new ByteArrayOutputStream();
        try (ZipOutputStream zos = new ZipOutputStream(baos)) {
            Files.list(sourceDir).forEach(file -> {
                try {
                    ZipEntry zipEntry = new ZipEntry(file.getFileName().toString());
                    zos.putNextEntry(zipEntry);
                    Files.copy(file, zos);
                    zos.closeEntry();
                } catch (IOException e) {
                   System.err.println("Erreur lors de l'ajout du fichier au ZIP : " + file.getFileName());
                }
            });
            String readmeContent = "Ce dossier contient les preuves de conformité générées le " + LocalDate.now() + ".\n"
                                 + "Le fichier .tsr est une réponse d'horodatage RFC 3161.\n"
                                 + "Le fichier .sha256 est l'empreinte numérique du PDF qui a été horodatée.";
            ZipEntry readmeEntry = new ZipEntry("README_recevabilite.txt");
            zos.putNextEntry(readmeEntry);
            zos.write(readmeContent.getBytes());
            zos.closeEntry();
        }
        return baos.toByteArray();
    }

    private static String toHexString(byte[] hash) {
        try (Formatter formatter = new Formatter()) {
            for (byte b : hash) {
                formatter.format("%02x", b);
            }
            return formatter.toString();
        }
    }

    public AuthResponse generateAuthResponse(String username) {

        String token = jwtService.generateToken(user);

                .collect(Collectors.toSet());

        // On utilise le constructeur complet, en passant un GroupSummary

                user.getId(),
                user.getUsername(),
                user.getEmail(),
                agentIds,
                user.getRole().name(),
                new GroupSummary(user.getCustomerGroup())
        );

        return new AuthResponse(true, "Connexion réussie", token, userInfo);
    }

    /**
     * Vérifie la validité d'un token JWT et retourne les informations de l'utilisateur.
     * C'est la méthode CORRIGÉE.
     */
    public AuthResponse verifyToken(String token) {
        // Étape 1: Extraire l'ID de l'utilisateur du token.
        final Long userId = jwtService.extractUserId(token);
        if (userId == null) {
            throw new RuntimeException("Token invalide: ID utilisateur manquant.");
        }

        // Étape 2: Charger les UserDetails correspondants à cet ID.
        // Notre UserDetailsServiceImpl a maintenant une méthode loadUserById.
        UserDetails userDetails = userDetailsService.loadUserById(userId);

        // Étape 3: Valider le token en le comparant avec les UserDetails.
        if (!jwtService.isTokenValid(token, userDetails)) {
            throw new RuntimeException("Token invalide ou expiré");
        }
        
        // Étape 4: Le token est valide. On récupère le vrai username et on génère la réponse.
        // Le "username" dans UserDetails est l'ID, donc on doit re-chercher le User.
        User user = userRepository.findById(userId)
            .orElseThrow(() -> new RuntimeException("Utilisateur non trouvé pour un token valide, ID: " + userId));

        // On peut maintenant réutiliser notre méthode de génération de réponse
        return generateAuthResponse(user.getUsername());
    }

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.lang.NonNull;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.security.core.userdetails.UserDetails;
import org.springframework.security.web.authentication.WebAuthenticationDetailsSource;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

@Component
public class JwtAuthenticationFilter extends OncePerRequestFilter {

    private final JwtService jwtService;
    private final UserDetailsServiceImpl userDetailsService;

    public JwtAuthenticationFilter(JwtService jwtService, UserDetailsServiceImpl userDetailsService) {
        this.jwtService = jwtService;
        this.userDetailsService = userDetailsService;
    }

    @Override
    protected void doFilterInternal(
            @NonNull HttpServletRequest request,
            @NonNull HttpServletResponse response,
            @NonNull FilterChain filterChain) throws ServletException, IOException {

        final String authHeader = request.getHeader("Authorization");
        if (authHeader == null || !authHeader.startsWith("Bearer ")) {
            filterChain.doFilter(request, response);
            return;
        }

        final String jwt = authHeader.substring(7);
        final Long userId;
        try {
            userId = jwtService.extractUserId(jwt);
        } catch (Exception e) {
            // Si le token est malformé, on arrête
            filterChain.doFilter(request, response);
            return;
        }

        if (userId != null && SecurityContextHolder.getContext().getAuthentication() == null) {
            UserDetails userDetails = this.userDetailsService.loadUserById(userId);
            if (jwtService.isTokenValid(jwt, userDetails)) {
                UsernamePasswordAuthenticationToken authToken = new UsernamePasswordAuthenticationToken(
                        userDetails, null, userDetails.getAuthorities()
                );
                authToken.setDetails(new WebAuthenticationDetailsSource().buildDetails(request));
                SecurityContextHolder.getContext().setAuthentication(authToken);
            }
        }
        filterChain.doFilter(request, response);
    }

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
public class SecurityConfig {

    private final JwtAuthenticationFilter jwtAuthFilter;
    private final UserDetailsServiceImpl userDetailsService;

    public SecurityConfig(JwtAuthenticationFilter jwtAuthFilter, UserDetailsServiceImpl userDetailsService) {
        this.jwtAuthFilter = jwtAuthFilter;
        this.userDetailsService = userDetailsService;
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
    
    @Bean
    public AuthenticationManager authenticationManager(AuthenticationConfiguration config) throws Exception {
        return config.getAuthenticationManager();
    }

    @Bean
    public AuthenticationProvider authenticationProvider() {
        DaoAuthenticationProvider authProvider = new DaoAuthenticationProvider();
        authProvider.setUserDetailsService(userDetailsService);
        authProvider.setPasswordEncoder(passwordEncoder());
        return authProvider;
    }

    /**
     * Chaîne de sécurité n°1 : Pour les APIs (/api/**)
     */
    @Bean
    @Order(1)
    public SecurityFilterChain apiSecurityFilterChain(HttpSecurity http) throws Exception {
        http
            .securityMatcher("/api/**")
            .csrf(AbstractHttpConfigurer::disable)
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/v1/auth/**").permitAll()
                .anyRequest().authenticated()
            )
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authenticationProvider(authenticationProvider())
            .addFilterBefore(jwtAuthFilter, UsernamePasswordAuthenticationFilter.class);
            
        return http.build();
    }

    /**
     * Chaîne de sécurité n°2 : Pour tout le reste
     */
    @Bean
    @Order(2)
    public SecurityFilterChain webSecurityFilterChain(HttpSecurity http) throws Exception {
        http
            // On désactive complètement le CSRF pour les routes non-API
            .csrf(AbstractHttpConfigurer::disable)
            .authorizeHttpRequests(auth -> auth
                // ON AUTORISE EXPLICITEMENT LA CONNEXION WEBSOCKET
                .requestMatchers("/ws-connect/**").permitAll()
                // Et on autorise tout le reste (frontend, swagger, h2)
                .anyRequest().permitAll()
            )
            .headers(headers -> headers.frameOptions(HeadersConfigurer.FrameOptionsConfig::sameOrigin)); // Pour H2

        return http.build();
    }

        this.authService = authService;
        this.authenticationManager = authenticationManager;
        this.userRepository = userRepository;
    }

    @PostMapping("/login")
    public ResponseEntity<AuthResponse> login(@RequestBody LoginRequest request) {
        try {
            Authentication authentication = authenticationManager.authenticate(
                new UsernamePasswordAuthenticationToken(request.getEmail(), request.getPassword())
            );

            // Le "principal" est maintenant l'ID de l'utilisateur sous forme de String
            String userId = authentication.getName();
            
            // On récupère le vrai username pour générer la réponse
            User user = userRepository.findById(Long.parseLong(userId))
                    .orElseThrow(() -> new RuntimeException("Utilisateur authentifié non trouvé en BDD, ID: " + userId));

            AuthResponse response = authService.generateAuthResponse(user.getUsername());
            return ResponseEntity.ok(response);
            
        } catch (Exception e) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(new AuthResponse(false, "Identifiants incorrects", null, null));
        }
    }
    
    // ... Le reste du contrôleur (verify, logout) est inchangé ...
    @GetMapping("/verify")
    public ResponseEntity<AuthResponse> verifyToken(@RequestHeader("Authorization") String authHeader) {
        try {
            String token = authHeader.replace("Bearer ", "");
            AuthResponse response = authService.verifyToken(token);
            return ResponseEntity.ok(response);
        } catch (Exception e) {
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED)
                    .body(new AuthResponse(false, "Token invalide", null, null));
        }
    }

    @PostMapping("/logout")
    public ResponseEntity<String> logout(@RequestHeader("Authorization") String authHeader) {
        try {
            String token = authHeader.replace("Bearer ", "");
            authService.logout(token);
            return ResponseEntity.ok("Déconnexion réussie");
        } catch (Exception e) {
            return ResponseEntity.badRequest().body("Erreur lors de la déconnexion");
        }
    }

    @Transactional
    public ResponseEntity<List<GroupResponse>> getAllGroups() {
        List<EnrichedAgentResponse> allEnrichedAgents = agentService.getAgentsForUser("superadmin");

        List<GroupResponse> groupResponses = groupService.getAllGroups().stream()
                .map(group -> {
                    Set<String> agentIdsInGroup = group.getAgents().stream()
                            .map(Agent::getId)
                            .collect(Collectors.toSet());
                    List<EnrichedAgentResponse> agentsForThisGroup = allEnrichedAgents.stream()
                            .filter(enrichedAgent -> agentIdsInGroup.contains(enrichedAgent.getId()))
                            .collect(Collectors.toList());
                    return new GroupResponse(group, agentsForThisGroup);
                })
                .collect(Collectors.toList());
                
        return ResponseEntity.ok(groupResponses);
    }

@Table(name = "managed_agents")
public class Agent {

    @Id
    @Column(name = "agent_id", length = 4)
    private String id;

    @Column(nullable = false)
    private String name;
    
    // --- NOUVEAU CHAMP ---
    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private AgentType type;

    
    @ManyToMany(mappedBy = "agents")
    @JsonIgnore // Évite les boucles infinies lors de la sérialisation JSON
    private Set<User> users = new HashSet<>();

    // --- Constructeurs ---
    public Agent() {
        // Constructeur vide requis par JPA
    }

    public Agent(String id, String name) {
        this.id = id;
        this.name = name;
        this.type = AgentType.UNKNOWN; // Type par défaut à la création simple
    }

    // --- Getters et Setters ---
    
    public AgentType getType() { return type; }
    public void setType(AgentType type) { this.type = type; }
    
    public String getId() {
        return id;
    }

    public void setId(String id) {
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Set<User> getUsers() {
        return users;
    }

    public void setUsers(Set<User> users) {
        this.users = users;
    }
}// src/main/java/xns/x3rsi/model/User.java
package xns.x3rsi.model;

import com.fasterxml.jackson.annotation.JsonIgnore;
import jakarta.persistence.*;
import xns.x3rsi.model.enums.Role; // <-- Importer notre enum

import java.util.HashSet;
import java.util.Set;

@Entity
@Table(name = "app_users")
public class User {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(unique = true, nullable = false, length = 50)
    private String username;

    @Column(nullable = false)
    @JsonIgnore
    private String password;

    @Column(unique = true, nullable = false)
    private String email;
    
    private boolean enabled = true;

    // --- NOUVEAU CHAMP : RÔLE ---
    @Enumerated(EnumType.STRING) // Stocke le nom de l'enum ("ROLE_USER") dans la BDD
    @Column(nullable = false)
    private Role role;

    // --- NOUVELLE RELATION : APPARTENANCE À UN GROUPE ---
    // Un utilisateur appartient à un seul groupe.
    // Pour les SUPER_ADMIN, ce champ peut être null.
    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "group_id")
    @JsonIgnore // Pour éviter les boucles
    private CustomerGroup customerGroup;

    // --- RELATION MODIFIÉE : AGENTS ---
    // Cette relation ManyToMany est maintenant principalement pour le ROLE_USER
    // qui a un lien direct avec son agent.
    @ManyToMany(fetch = FetchType.LAZY)
    @JoinTable(
        name = "user_agents", 
        joinColumns = @JoinColumn(name = "user_id"), 
        inverseJoinColumns = @JoinColumn(name = "agent_id")
    )
    @JsonIgnore
    private Set<Agent> agents = new HashSet<>();

    // --- Constructeurs ---
    public User() {}

    // On met à jour le constructeur
    public User(String username, String email, String password, Role role) {
        this.username = username;
        this.email = email;
        this.password = password;
        this.role = role;
    }

    // --- Getters et Setters (Ajouter ceux pour les nouveaux champs) ---
    
    // ... (getters/setters existants pour id, username, password, email, enabled, agents)
    
    public Long getId() { return id; }
    public void setId(Long id) { this.id = id; }
    public String getUsername() { return username; }
    public void setUsername(String username) { this.username = username; }
    public String getPassword() { return password; }
    public void setPassword(String password) { this.password = password; }
    public String getEmail() { return email; }
    public void setEmail(String email) { this.email = email; }
    public boolean isEnabled() { return enabled; }
    public void setEnabled(boolean enabled) { this.enabled = enabled; }
    public Set<Agent> getAgents() { return agents; }
    public void setAgents(Set<Agent> agents) { this.agents = agents; }
    
    // Nouveaux getters/setters
    public Role getRole() { return role; }
    public void setRole(Role role) { this.role = role; }
    public CustomerGroup getCustomerGroup() { return customerGroup; }
    public void setCustomerGroup(CustomerGroup customerGroup) { this.customerGroup = customerGroup; }
}
